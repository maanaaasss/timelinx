## Install

```bash
npm install @timelinx/core @timelinx/react @timelinx/ui
```

## Create a Timeline

```ts
import {
  createTimeline,
  createTimelineState,
  createTrack,
  createClip,
  dispatch,
  toFrame,
  frameRate,
} from '@timelinx/core';

// Create a 10-second timeline at 30fps
const timeline = createTimeline({
  id: 'tl-1',
  name: 'My Edit',
  fps: frameRate(30),
  duration: toFrame(3000),
});

const state = createTimelineState({ timeline });
```

## Add a Track and Clip

```ts
const track = createTrack({ id: 'v1', name: 'V1', type: 'video', clips: [] });

const result = dispatch(state, {
  id: 'setup',
  label: 'Add track and clip',
  timestamp: Date.now(),
  operations: [
    { type: 'ADD_TRACK', track },
    { type: 'INSERT_CLIP', trackId: track.id, clip: createClip({
      id: 'clip-1',
      assetId: 'my-video',
      trackId: 'v1',
      timelineStart: toFrame(0),
      timelineEnd: toFrame(300),
      mediaIn: toFrame(0),
      mediaOut: toFrame(300),
    })},
  ],
});

if (!result.accepted) {
  throw new Error(`${result.reason}: ${result.message}`);
}

const ready = result.nextState;
```

## Render the Editor

```tsx
'use client';

import { TimelineEditor } from '@timelinx/ui';

export default function Editor() {
  return <TimelineEditor initialState={ready} />;
}
```

That's it. You have a working timeline editor with tracks, clips, zoom, drag, undo/redo, and keyboard shortcuts.
